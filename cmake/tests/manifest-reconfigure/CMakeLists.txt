cmake_minimum_required(VERSION 3.20)
project(ManifestRegeneration NONE)
include("${AROS_MANIFEST_HELPER}")
aros_collect_mmake_manifests(_manifests)
set(_count 0)
if(EXISTS "${CMAKE_BINARY_DIR}/configure-count.txt")
  file(READ "${CMAKE_BINARY_DIR}/configure-count.txt" _count)
endif()
math(EXPR _count "${_count} + 1")
file(WRITE "${CMAKE_BINARY_DIR}/configure-count.txt" "${_count}")
set(_inventory "")
foreach(_manifest IN LISTS _manifests)
  file(RELATIVE_PATH _relative "${CMAKE_SOURCE_DIR}" "${_manifest}")
  file(SHA256 "${_manifest}" _hash)
  string(APPEND _inventory "${_relative}:${_hash}\n")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/configured.txt" "${_inventory}")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/built.txt"
  COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_BINARY_DIR}/configured.txt" "${CMAKE_BINARY_DIR}/built.txt"
  DEPENDS "${CMAKE_BINARY_DIR}/configured.txt" VERBATIM)
add_custom_target(manifest-check DEPENDS "${CMAKE_BINARY_DIR}/built.txt")
