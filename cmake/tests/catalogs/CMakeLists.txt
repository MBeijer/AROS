cmake_minimum_required(VERSION 3.20)
project(AROSCatalogFixture C)
include("${AROS_TEST_SOURCE}/cmake/AROSModule.cmake")
include("${AROS_TEST_SOURCE}/cmake/AROSProgramTargets.cmake")
set(AROS_TARGET linux-x86_64)
set(AROS_TARGET_FAMILY unix)
set(AROS_LEGACY_BUILD_DIR "${CMAKE_SOURCE_DIR}/legacy")
set(AROS_NATIVE_OUTPUT_ROOT "${CMAKE_BINARY_DIR}/output")
set(AROS_NATIVE_SYSTEM_DIR "${AROS_NATIVE_OUTPUT_ROOT}/AROS")
set(_AROS_MMAKE_VAR_AROS_CATALOGS "$(AROSDIR)/Locale/Catalogs")
set(AROS_CROSSTOOLS_TARGET_CPU test)
set(AROS_NATIVE_TOOLCHAIN_INSTALL_DIR "${CMAKE_BINARY_DIR}/toolchain")
set(AROS_NATIVE_DEVELOPER_DIR "${AROS_NATIVE_SYSTEM_DIR}/Development")
set(AROS_NATIVE_PUBLIC_LIB_DIR "${AROS_NATIVE_DEVELOPER_DIR}/lib")
set(AROS_NATIVE_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/include")
file(MAKE_DIRECTORY "${AROS_NATIVE_TOOLCHAIN_INSTALL_DIR}")
file(CREATE_LINK "${CMAKE_C_COMPILER}" "${AROS_NATIVE_TOOLCHAIN_INSTALL_DIR}/test-aros-gcc" SYMBOLIC)
add_executable(flexcat "${CMAKE_SOURCE_DIR}/flexcat.c")
set_target_properties(flexcat PROPERTIES
  AROS_FLEXCAT_SOURCE_DESCRIPTION_DIR "${CMAKE_SOURCE_DIR}/sd")
aros_collect_mmake_manifests(AROS_MMAKE_MANIFESTS)
_aros_register_program_catalogs(fixture-catalogs _target)
if(NOT _target)
  message(FATAL_ERROR "Missing fixture-catalogs producer")
endif()
_aros_register_program_catalogs(fixture-catalogs _again)
if(NOT _target STREQUAL _again)
  message(FATAL_ERROR "Repeated dependency registration changed the producer")
endif()
add_custom_target(catalogs DEPENDS "${_target}")
if(TEST_SECOND_PRODUCER)
  _aros_register_program_catalogs(second-catalogs _second)
endif()
if(TEST_PROGRAM)
  aros_add_mmake_programs(consumer MMAKE_NAME fixture-program)
  aros_finalize_program_dependencies()
endif()
