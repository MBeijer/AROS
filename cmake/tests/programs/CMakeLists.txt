cmake_minimum_required(VERSION 3.20)
project(AROSProgramFixture C)
include("${AROS_TEST_SOURCE}/cmake/AROSModule.cmake")
include("${AROS_TEST_SOURCE}/cmake/AROSManifests.cmake")
include("${AROS_TEST_SOURCE}/cmake/AROSProgramTargets.cmake")
set(AROS_TARGET linux-x86_64)
set(AROS_TARGET_FAMILY unix)
set(AROS_CROSSTOOLS_TARGET_CPU test)
set(AROS_LEGACY_BUILD_DIR "${CMAKE_SOURCE_DIR}/config-snapshot")
set(AROS_NATIVE_TOOLCHAIN_INSTALL_DIR "${CMAKE_BINARY_DIR}/toolchain")
set(AROS_NATIVE_OUTPUT_ROOT "${CMAKE_BINARY_DIR}/output")
set(AROS_NATIVE_SYSTEM_DIR "${AROS_NATIVE_OUTPUT_ROOT}/AROS")
set(AROS_NATIVE_DEVELOPER_DIR "${AROS_NATIVE_SYSTEM_DIR}/Development")
set(AROS_NATIVE_PUBLIC_LIB_DIR "${AROS_NATIVE_DEVELOPER_DIR}/lib")
set(AROS_NATIVE_BUILD_SDKS_DIR "${AROS_NATIVE_OUTPUT_ROOT}/gen/buildsdks")
set(AROS_NATIVE_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/include")
if(AROS_TEST_INCLUDE_DIR)
  set(AROS_NATIVE_INCLUDE_DIR "${AROS_TEST_INCLUDE_DIR}")
endif()
file(MAKE_DIRECTORY "${AROS_NATIVE_TOOLCHAIN_INSTALL_DIR}")
option(AROS_TEST_INSTALL_COMPILER "Install the fixture compiler immediately" ON)
if(AROS_TEST_INSTALL_COMPILER)
  file(CREATE_LINK "${CMAKE_C_COMPILER}" "${AROS_NATIVE_TOOLCHAIN_INSTALL_DIR}/test-aros-gcc" SYMBOLIC)
  file(CREATE_LINK "${CMAKE_STRIP}" "${AROS_NATIVE_TOOLCHAIN_INSTALL_DIR}/test-aros-strip" SYMBOLIC)
endif()
add_library(fixture-support STATIC "${CMAKE_SOURCE_DIR}/support.c")
set_target_properties(fixture-support PROPERTIES OUTPUT_NAME fixture
  ARCHIVE_OUTPUT_DIRECTORY "${AROS_NATIVE_PUBLIC_LIB_DIR}"
  AROS_LINKLIB_OUTPUT "${AROS_NATIVE_PUBLIC_LIB_DIR}/libfixture.a")
set_property(GLOBAL PROPERTY AROS_REGISTERED_LINKLIB_TARGETS fixture-support)
_aros_sanitize_property_key(fixture _lib_key)
set_property(GLOBAL PROPERTY "AROS_REGISTERED_LINKLIB_TARGET_${_lib_key}" fixture-support)
set_property(GLOBAL PROPERTY "AROS_REGISTERED_LINKLIB_OUTPUT_${_lib_key}"
  "${AROS_NATIVE_PUBLIC_LIB_DIR}/libfixture.a")
add_custom_target(unrelated-interface
  COMMAND "${CMAKE_COMMAND}" -E echo "Unrelated SDK interface must not build"
  COMMAND "${CMAKE_COMMAND}" -E false)
set_target_properties(unrelated-interface PROPERTIES
  AROS_MODULE_INTERFACE_STAMP "${CMAKE_BINARY_DIR}/unrelated.stamp")
set_property(GLOBAL PROPERTY AROS_REGISTERED_MODULE_INTERFACE_TARGETS unrelated-interface)
add_custom_target(aros-sdk-interfaces-native DEPENDS unrelated-interface)
set(_extra_header "${AROS_NATIVE_INCLUDE_DIR}/proto/extra.h")
file(MAKE_DIRECTORY "${AROS_NATIVE_INCLUDE_DIR}/proto")
add_custom_command(OUTPUT "${_extra_header}"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_SOURCE_DIR}/extra.h" "${_extra_header}"
  DEPENDS "${CMAKE_SOURCE_DIR}/extra.h" VERBATIM)
add_custom_target(extra-interface DEPENDS "${_extra_header}")
set_target_properties(extra-interface PROPERTIES AROS_MODULE_INTERFACE_STAMP "${_extra_header}")
_aros_sanitize_property_key(extra _extra_key)
set_property(GLOBAL PROPERTY "AROS_MODULE_NAME_INTERFACE_TARGET_${_extra_key}" extra-interface)
if(EXISTS "${CMAKE_SOURCE_DIR}/test-icon.c")
  add_executable(ilbmtoicon "${CMAKE_SOURCE_DIR}/test-icon.c")
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/runtime.txt" AND NOT AROS_TEST_NO_RUNTIME_PRODUCER)
  set(_runtime "${AROS_NATIVE_SYSTEM_DIR}/Libs/fixture.library")
  add_custom_command(OUTPUT "${_runtime}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${AROS_NATIVE_SYSTEM_DIR}/Libs"
    COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_SOURCE_DIR}/runtime.txt" "${_runtime}"
    DEPENDS "${CMAKE_SOURCE_DIR}/runtime.txt" VERBATIM)
  add_custom_target(fixture-runtime-native DEPENDS "${_runtime}")
  _aros_register_mmake_runtime_target(fixture-runtime fixture-runtime-native)
endif()
aros_collect_mmake_manifests(AROS_MMAKE_MANIFESTS)
if(AROS_TEST_AUTO_PROGRAMS)
  aros_add_program_graph(fixture-programs ROOT "${AROS_TEST_PROGRAM_ROOT}")
  get_target_property(_programs fixture-programs AROS_PROGRAM_GRAPH_RULES)
  file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/programs.txt" CONTENT "${_programs}\n" @ONLY)
else()
  aros_add_mmake_programs(fixture-programs MMAKE_NAME fixture-commands)
endif()
aros_finalize_program_dependencies()
if(AROS_TEST_AUTO_PROGRAMS)
  get_property(_registered GLOBAL PROPERTY AROS_NATIVE_PROGRAM_TARGETS)
  set(_orders "")
  foreach(_registered_target IN LISTS _registered)
    get_target_property(_identity "${_registered_target}" AROS_PROGRAM_MMAKE)
    if(_identity STREQUAL "commands")
      get_target_property(_children "${_registered_target}" AROS_PROGRAM_TARGETS)
      foreach(_child IN LISTS _children)
        get_target_property(_order "${_child}" MANUALLY_ADDED_DEPENDENCIES)
        string(APPEND _orders "${_child}: ${_order}\n")
      endforeach()
    endif()
  endforeach()
  file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/program-order.txt" CONTENT "${_orders}" @ONLY)
endif()
