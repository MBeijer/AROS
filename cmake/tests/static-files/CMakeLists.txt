cmake_minimum_required(VERSION 3.20)
project(AROSStaticFileFixture NONE)
include("${AROS_CMAKE_DIR}/AROSModule.cmake")
include("${AROS_CMAKE_DIR}/AROSManifests.cmake")
include("${AROS_CMAKE_DIR}/AROSMmakeRules.cmake")
set(AROS_TARGET linux-x86_64)
set(AROS_TARGET_FAMILY unix)
set(AROS_NATIVE_SYSTEM_DIR "${CMAKE_BINARY_DIR}/runtime")
aros_collect_mmake_manifests(_manifests)
if(AROS_STATIC_GRAPH OR AROS_STATIC_LOCAL)
  set(AROS_MMAKE_MANIFESTS "${_manifests}")
  include("${AROS_CMAKE_DIR}/AROSStaticTargets.cmake")
  if(AROS_STATIC_LOCAL)
    aros_add_mmake_local_static_target(static-files MMAKE_NAME example)
  else()
    aros_add_mmake_static_target(static-files MMAKE_NAME example)
  endif()
else()
  aros_add_mmake_static_files(static-files MODULE_PATH data MMAKE_NAME example)
endif()
